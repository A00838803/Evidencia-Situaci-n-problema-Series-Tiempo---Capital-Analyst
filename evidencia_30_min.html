# Random Walk Test (Variance Ratio Test)
def random_walk_test(series, name):
    print(f"\nRandom Walk Test for {name}:")
    diffs = series.diff().dropna()
    var_original = np.var(series)
    var_diff = np.var(diffs)
    ratio = var_original / (len(series) * var_diff)
    print(f"Variance Ratio: {ratio:.4f}")
    print("Interpretation:")
    if abs(ratio - 1) < 0.1:
        print(f"  - Variance ratio ≈ 1: {name} behaves like a random walk")
    else:
        print(f"  - Variance ratio ≠ 1: {name} may not be a pure random walk")

random_walk_test(pltr_close, "PLTR")
random_walk_test(bsx_close, "BSX")
random_walk_test(aapl_close, "AAPL")

# Unit Root Tests (ADF and KPSS)
def unit_root_tests(series, name):
    print(f"\nUnit Root Tests for {name}:")
    # ADF Test
    adf_result = adfuller(series)
    print("ADF Test:")
    print(f'ADF Statistic: {adf_result[0]:.4f}')
    print(f'p-value: {adf_result[1]:.4f}')
    print(f'Critical Values: {adf_result[4]}')
    print("Interpretation:")
    if adf_result[1] < 0.05:
        print(f"  - p-value < 0.05: Reject null hypothesis - {name} is stationary")
    else:
        print(f"  - p-value >= 0.05: Fail to reject null - {name} may be non-stationary")
    # KPSS Test
    kpss_result = kpss(series, regression='c')
    print("\nKPSS Test:")
    print(f'KPSS Statistic: {kpss_result[0]:.4f}')
    print(f'p-value: {kpss_result[1]:.4f}')
    print(f'Critical Values: {kpss_result[3]}')
    print("Interpretation:")
    if kpss_result[1] < 0.05:
        print(f"  - p-value < 0.05: Reject null hypothesis - {name} is non-stationary")
    else:
        print(f"  - p-value >= 0.05: Fail to reject null - {name} may be stationary")

unit_root_tests(pltr_close, "PLTR")
unit_root_tests(bsx_close, "BSX")
unit_root_tests(aapl_close, "AAPL")

# Function to fit and evaluate AR, ARMA, and ARIMA models
def fit_time_series_models(series, name, max_p=3, max_q=3):
    print(f"\nFitting Time Series Models for {name}:")

    # Store results for comparison
    models_results = []

    # Fit AR models (ARIMA with q=0, d=0)
    best_ar_aic = float('inf')
    best_ar_order = None
    for p in range(1, max_p + 1):
        try:
            model = ARIMA(series, order=(p, 0, 0))
            results = model.fit()
            models_results.append(('AR', (p, 0, 0), results.aic))
            if results.aic < best_ar_aic:
                best_ar_aic = results.aic
                best_ar_order = (p, 0, 0)
        except:
            continue

    # Fit ARMA models (ARIMA with p>0, q>0, d=0)
    best_arma_aic = float('inf')
    best_arma_order = None
    for p in range(1, max_p + 1):
        for q in range(1, max_q + 1):
            try:
                model = ARIMA(series, order=(p, 0, q))
                results = model.fit()
                models_results.append(('ARMA', (p, 0, q), results.aic))
                if results.aic < best_arma_aic:
                    best_arma_aic = results.aic
                    best_arma_order = (p, 0, q)
            except:
                continue

    # Fit ARIMA models (d=1 to account for differencing)
    best_arima_aic = float('inf')
    best_arima_order = None
    for p in range(max_p + 1):
        for q in range(max_q + 1):
            try:
                model = ARIMA(series, order=(p, 1, q))
                results = model.fit()
                models_results.append(('ARIMA', (p, 1, q), results.aic))
                if results.aic < best_arima_aic:
                    best_arima_aic = results.aic
                    best_arima_order = (p, 1, q)
            except:
                continue

    # Print model comparison
    print(f"\nModel Comparison for {name}:")
    print("Model Type | Order | AIC")
    print("-" * 30)
    for model_type, order, aic in models_results:
        print(f"{model_type:<10} | {order} | {aic:.2f}")

    # Select the best overall model
    best_model = min(models_results, key=lambda x: x[2])
    best_model_type, best_order, best_aic = best_model
    print(f"\nBest Model for {name}:")
    print(f"Type: {best_model_type}")
    print(f"Order: {best_order}")
    print(f"AIC: {best_aic:.2f}")
    print("Interpretation:")
    if best_model_type == 'AR':
        print(f"  - AR({best_order[0]}): {best_order[0]} autoregressive term(s)")
    elif best_model_type == 'ARMA':
        print(f"  - ARMA({best_order[0]},{best_order[2]}): {best_order[0]} autoregressive term(s), {best_order[2]} moving average term(s)")
    elif best_model_type == 'ARIMA':
        print(f"  - ARIMA({best_order[0]},1,{best_order[2]}): {best_order[0]} autoregressive term(s), 1 differencing, {best_order[2]} moving average term(s)")

    return best_model_type, best_order

# Cointegration Test
def cointegration_test(df):
    result = coint_johansen(df, det_order=0, k_ar_diff=1)
    print("\nJohansen Cointegration Test (PLTR, BSX, AAPL):")
    print(f"Trace statistic: {result.lr1}")
    print(f"Critical values (90%, 95%, 99%): {result.cvt}")
    print("Interpretation:")
    for i in range(len(result.lr1)):
        if result.lr1[i] > result.cvt[i, 1]:
            print(f"  - r = {i}: Cointegration exists at 95% confidence level")
            print(f"    Trace statistic ({result.lr1[i]:.2f}) > 95% critical value ({result.cvt[i, 1]:.2f})")
        else:
            print(f"  - r = {i}: No cointegration at 95% confidence level")
            print(f"    Trace statistic ({result.lr1[i]:.2f}) <= 95% critical value ({result.cvt[i, 1]:.2f})")
    if result.lr1[0] > result.cvt[0, 1]:
        print("Conclusion: PLTR, BSX, and AAPL are cointegrated - they share a long-run equilibrium relationship")
    else:
        print("Conclusion: No evidence of cointegration among PLTR, BSX, and AAPL")

# Detailed forecast plot with confidence intervals
def plot_forecast(model, series, name, steps=30, is_diff_series=False):
    forecast_obj = model.get_forecast(steps=steps)
    forecast = forecast_obj.predicted_mean
    conf_int = forecast_obj.conf_int()
    forecast_index = range(len(series), len(series) + steps)
    plt.figure(figsize=(12, 6))
    plt.plot(series, label=f'{name} Historical {"(Differenced)" if is_diff_series else ""}')
    plt.plot(forecast_index, forecast, label='Forecast', color='red')
    plt.fill_between(forecast_index,
                     conf_int.iloc[:, 0],
                     conf_int.iloc[:, 1],
                     color='pink',
                     alpha=0.3,
                     label='95% Confidence Interval')
    plt.title(f'{name} Price Forecast (30-min)')
    plt.legend()
    plt.show()
    # Forecast interpretation
    last_value = series.iloc[-1]
    mean_forecast = forecast.mean()
    print(f"\nForecast Interpretation for {name}:")
    print(f"Last observed value: {last_value:.2f}")
    print(f"Average forecast value: {mean_forecast:.2f}")
    print(f"Forecast change: {mean_forecast - last_value:.2f}")
    if mean_forecast > last_value:
        print("Trend: Upward forecast trend")
    elif mean_forecast < last_value:
        print("Trend: Downward forecast trend")
    else:
        print("Trend: Flat forecast trend")
    print(f"95% CI range at period {steps}: [{conf_int.iloc[-1, 0]:.2f}, {conf_int.iloc[-1, 1]:.2f}]")

# Generate detailed forecast plots and interpretations
